const { onCall, HttpsError } = require("firebase-functions/v2/https"); const { defineSecret } = require("firebase-functions/params"); const OpenAI = require("openai"); const OPENAI_API_KEY = defineSecret("OPENAI_API_KEY"); exports.askBestie = onCall( { region: "europe-west1", secrets: [OPENAI_API_KEY], timeoutSeconds: 60 }, async (request) => { if (!request.auth) { throw new HttpsError("unauthenticated", "Log in first."); } const messages = request.data && request.data.messages; if (!Array.isArray(messages) || messages.length === 0 || messages.length > 12) { throw new HttpsError("invalid-argument", "Send a short conversation."); } const safeMessages = messages.map((message) => { if (!message || !["user", "assistant"].includes(message.role) || typeof message.content !== "string") { throw new HttpsError("invalid-argument", "Invalid chat message."); } return { role: message.role, content: message.content.slice(0, 2000) }; }); try { const client = new OpenAI({ apiKey: OPENAI_API_KEY.value() }); const response = await client.responses.create({ model: "gpt-5.5", store: false, instructions: [ "You are Virtual Bestie's friendly, thoughtful chat companion for a teen audience.", "Respond to the actual message and use the recent conversation for context. Do not repeat generic greetings or canned replies.", "Be warm, natural, specific, and honest. If you are unsure, say so instead of inventing facts.", "Use clear age-appropriate language. Do not produce sexual content, self-harm instructions, dangerous activity instructions, or guidance for obtaining restricted items.", "Do not ask for passwords, home addresses, phone numbers, or other sensitive personal information.", "Do not claim to be a human, a real-life friend, or a substitute for trusted adults or professional help.", "If the user may be in immediate danger or is dealing with something serious, encourage them to contact a trusted adult or local emergency help.", "Answer in the language the user uses. Keep replies concise unless detail is useful." ].join(" "), input: safeMessages, max_output_tokens: 500 }); const answer = (response.output_text || "").trim(); if (!answer) { throw new Error("The model returned an empty answer."); } return { answer }; } catch (error) { console.error("OpenAI request failed:", error.message); throw new HttpsError("internal", "The chat service is temporarily unavailable."); } } );
